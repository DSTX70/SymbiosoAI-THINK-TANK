// ————— Helper: client-side script injected into pages —————
const CLIENT_JS = `
  (function(){
    const $ = (s,r=document)=>r.querySelector(s);
    const setText = (el, html) => el && (el.innerHTML = html);

    function wireSimple() {
      const btn = document.getElementById("btn-simple-run");
      if(!btn) return;
      btn.addEventListener("click", async () => {
        const prompt = $("#simple-prompt").value.trim();
        const requireCitations = $("#simple-cite").checked;
        const factCheck = $("#simple-fact").checked;

        setText($("#consensus"), "<em class='muted'>Processing…</em>");
        setText($("#dissents"), "");
        setText($("#unresolved"), "");

        try {
          const res = await fetch("/api/think", {
            method:"POST",
            headers:{ "Content-Type":"application/json" },
            body:JSON.stringify({
              prompt, mode:"simple",
              require_citations: requireCitations,
              enable_fact_check: factCheck,
              temperature: 0.2
            })
          });
          const data = await res.json();
          renderResults(data, "simple");
        } catch (e) {
          setText($("#consensus"), "<span style='color:#b91c1c'>Error processing request.</span>");
        }
      });
    }

    function wireGuided() {
      const btn = document.getElementById("btn-guided-run");
      if(!btn) return;
      btn.addEventListener("click", async () => {
        const prompt = $("#guided-prompt").value.trim();
        const selection = $("#sel-mode").value;
        const length = $("#sel-length").value;
        const rounds = Number($("#inp-rounds").value || 3);
        const format = $("#sel-format").value;
        const reqEv = $("#chk-evidence").checked;
        const reqCnt = $("#chk-counter").checked;
        const cite = $("#chk-cite").checked;
        const fact = $("#chk-fact").checked;
        const minsrc = Number($("#inp-minsrc").value || 0);

        setText($("#g-consensus"), "<em class='muted'>Processing…</em>");
        setText($("#g-dissents"), "");
        setText($("#g-unresolved"), "");

        try {
          const res = await fetch("/api/think", {
            method:"POST",
            headers:{ "Content-Type":"application/json" },
            body:JSON.stringify({
              prompt, mode:"guided",
              selection_mode: selection,
              response_length: length,
              turns: rounds,
              debate_format: format,
              require_evidence: reqEv,
              require_counterarguments: reqCnt,
              verification: { require_citations: cite, fact_check: fact, min_sources: minsrc },
              temperature: 0.2
            })
          });
          const data = await res.json();
          renderResults(data, "guided");
        } catch (e) {
          setText($("#g-consensus"), "<span style='color:#b91c1c'>Error processing request.</span>");
        }
      });
    }

    function renderResults(data, mode){
      const consensusEl = mode === "simple" ? $("#consensus") : $("#g-consensus");
      const dissentsEl = mode === "simple" ? $("#dissents") : $("#g-dissents");
      const unresolvedEl = mode === "simple" ? $("#unresolved") : $("#g-unresolved");

      setText(consensusEl, data.consensus ? data.consensus : "<em class='muted'>No consensus.</em>");
      setText(dissentsEl, "<ul class='muted'>" + (data.dissents||[]).map(d => "<li>"+(d.position||d)+"</li>").join("") + "</ul>");
      setText(unresolvedEl, "<ul class='muted'>" + (data.unresolved||[]).map(u => "<li>"+u+"</li>").join("") + "</ul>");
    }

    wireSimple();
    wireGuided();
  })();
`;

// ————— API: stub /api/think (structured like your schema) —————
app.post("/api/think", async (req, res) => {
  const { prompt = "", mode = "simple" } = req.body || {};
  // Simulate processing latency
  await new Promise(r => setTimeout(r, 900));

  const consensus =
    mode === "guided"
      ? `Guided consensus for “${prompt || "your prompt"}”. Settings applied.`
      : `Simple consensus for “${prompt || "your prompt"}”.`;

  res.json({
    consensus,
    dissents: [
      "Alternative schedule: 9/80 can improve coverage",
      "Risk of hidden overtime; mitigation needed"
    ],
    unresolved: [
      "Define SLA guardbands under compressed weeks",
      "Identify roles requiring on-site presence"
    ],
    telemetry: { avg_ms: 780, quality: 4.7, tps: 21 }
  });
});
