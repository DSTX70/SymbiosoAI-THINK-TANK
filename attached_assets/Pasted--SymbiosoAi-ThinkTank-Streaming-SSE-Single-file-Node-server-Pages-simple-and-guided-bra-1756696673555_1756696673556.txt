// Injected client JS: connects to SSE and updates UI live
const CLIENT_JS_SSE = `
(function(){
  const $ = (s,r=document)=>r.querySelector(s);
  const set = (el, html) => el && (el.innerHTML = html);

  function connect(url, handlers){
    const es = new EventSource(url);
    const safe = (ev, fn) => es.addEventListener(ev, e => {
      try { fn(JSON.parse(e.data)); } catch { /* ignore */ }
    });
    safe("ready", d => handlers.onReady && handlers.onReady(d));
    safe("progress", d => handlers.onProgress && handlers.onProgress(d));
    safe("provider", d => handlers.onProvider && handlers.onProvider(d));
    safe("final", d => { handlers.onFinal && handlers.onFinal(d); es.close(); });
    es.onerror = () => { handlers.onError && handlers.onError("stream error"); es.close(); };
    return es;
  }

  function urlWithParams(path, params){
    const usp = new URLSearchParams(params);
    return path + "?" + usp.toString();
  }

  // Simple
  const btnS = $("#btn-simple-run");
  if(btnS){
    btnS.addEventListener("click", ()=>{
      const url = urlWithParams("/api/think/stream", {
        prompt: $("#simple-prompt").value.trim(),
        mode: "simple",
        require_citations: $("#simple-cite").checked ? "1" : "0",
        enable_fact_check: $("#simple-fact").checked ? "1" : "0",
        temperature: "0.2"
      });

      set($("#simple-consensus"), "<em class='muted'>Connecting…</em>");
      set($("#simple-dissents"), ""); set($("#simple-unresolved"), "");
      connect(url, {
        onProgress: d => set($("#simple-consensus"), "<em class='muted'>Processing… "+(d.pct||0)+"%</em>"),
        onProvider: d => { /* could show provider partials here */ },
        onFinal: d => {
          set($("#simple-consensus"), d.consensus || "<em class='muted'>No consensus.</em>");
          set($("#simple-dissents"), "<ul class='muted'>"+(d.dissents||[]).map(x=>"<li>"+(x.position||x)+"</li>").join("")+"</ul>");
          set($("#simple-unresolved"), "<ul class='muted'>"+(d.unresolved||[]).map(u=>"<li>"+u+"</li>").join("")+"</ul>");
          set($("#simple-telemetry"), "<p class='muted'>Avg Response: "+(d.telemetry?.avg_ms||"—")+"ms</p><p class='muted'>Quality: "+(d.telemetry?.quality||"—")+"</p><p class='muted'>Tokens/sec: "+(d.telemetry?.tps||"—")+"</p>");

      });
    });
  }

  // Guided
  const btnG = $("#btn-guided-run");
  if(btnG){
    btnG.addEventListener("click", ()=>{
      const url = urlWithParams("/api/think/stream", {

        fact: $("#chk-fact").checked ? "1" : "0",

      });

      connect(url, {
        onProgress: d => set($("#guided-consensus"), "<em class='muted'>Processing… "+(d.pct||0)+"%</em>"),

// ---------------- Utility ----------------
function withTimeout(promise, ms, label="request"){
  return Promise.race([
    promise,
    new Promise((_,rej)=>setTimeout(()=>rej(new Error(`Timeout after ${ms}ms (${label})`)), ms))
  ]);
}

// Call OpenAI / Anthropic with JSON-like instruction (not token streaming here — we stream SSE progress + provider milestones)
async function callOpenAI(prompt, sys, temperature){
  if(!openai) throw new Error("OpenAI not configured");
  const res = await withTimeout(
    openai.chat.completions.create({
      model: OPENAI_MODEL,
      temperature,
      messages:[
        { role:"system", content: sys },
        { role:"user", content: `${prompt}\nRespond as JSON: {"consensus": string, "points": string[]}` }
      ],
      response_format: { type: "json_object" }
    }),
    REQUEST_TIMEOUT_MS,
    "openai"
  );
  const content = res.choices?.[0]?.message?.content || "{}";
  return JSON.parse(content);
}
async function callAnthropic(prompt, sys, temperature){
  if(!anthropic) throw new Error("Anthropic not configured");
  const msg = await withTimeout(
    anthropic.messages.create({
      model: CLAUDE_MODEL,
      max_tokens: 800,
      temperature,
      system: sys,
      messages:[{ role:"user", content: `${prompt}\nRespond as JSON: {"consensus": string, "points": string[]}` }]
    }),
    REQUEST_TIMEOUT_MS,
    "anthropic"
  );
  const text = msg.content?.[0]?.text || "{}";
  return JSON.parse(text);
}

function mergeResults(oai, claude){
  const consensus = [oai?.consensus, claude?.consensus].filter(Boolean).join(" / ") || "No consensus.";
  const points = [].concat(oai?.points||[], claude?.points||[]);
  const dissents = points.slice(0, 3).map(p => ({ position: p }));
  const unresolved = ["Clarify assumptions", "Define success metrics"];
  return { consensus, dissents, unresolved };
}

// ---------------- SSE Streaming ----------------
app.get("/api/think/stream", async (req, res) => {
  // headers for SSE
  res.set({
    "Content-Type": "text/event-stream",
    "Cache-Control": "no-cache, no-transform",
    "Connection": "keep-alive",
    "X-Accel-Buffering": "no"
  });
  res.flushHeaders && res.flushHeaders();

  const send = (event, data) => {
    res.write(`event: ${event}\n`);
    res.write(`data: ${JSON.stringify(data)}\n\n`);
  };

  // keep-alive ping
  const ping = setInterval(() => res.write(": ping\n\n"), 15000);

  try {
    const {
      prompt = "",
      mode = "simple",
      temperature = "0.2",
      selection_mode, response_length, turns, debate_format,
      require_evidence, require_counterarguments, cite, fact, minsrc
    } = req.query;

    if (!prompt) {
      send("final", { consensus: "Please enter a prompt.", dissents: [], unresolved: [], telemetry: { avg_ms: 0 } });
      clearInterval(ping); res.end(); return;
    }

    send("ready", { ok: true });
    // progress ticker (UI-friendly 5%..92%)
    let pct = 5;
    const prog = setInterval(() => { pct = Math.min(92, pct + 3); send("progress", { pct }); }, 600);

    const start = Date.now();
    let oai = null, claude = null;
    const sysAnalyst = mode === "guided"
      ? "You are the 'Analyst' persona in a multi-model think tank. Return JSON."
      : "You are a precise analyst. Return JSON.";
    const sysCritic = mode === "guided"
      ? "You are the 'Critic' persona in a multi-model think tank. Return JSON."
      : "You are a critical reviewer. Return JSON.";

    const tasks = [];

    if (ENABLE_OPENAI) {
      tasks.push(
        callOpenAI(String(prompt), sysAnalyst, Number(temperature))
          .then(r => { oai = r; send("provider", { provider: "openai", status: "done" }); })
          .catch(e => { send("provider", { provider: "openai", status: "error", message: e.message }); })
      );
    }
    if (ENABLE_ANTHROPIC) {
      tasks.push(
        callAnthropic(String(prompt), sysCritic, Number(temperature))
          .then(r => { claude = r; send("provider", { provider: "anthropic", status: "done" }); })
          .catch(e => { send("provider", { provider: "anthropic", status: "error", message: e.message }); })
      );
    }

    if (tasks.length === 0) {
      clearInterval(prog);
      const elapsed = Date.now() - start;
      send("final", {
        consensus: "No AI providers configured (set OPENAI_API_KEY or ANTHROPIC_API_KEY).",
        dissents: [], unresolved: [], telemetry: { avg_ms: elapsed, quality: 0, tps: 0 }
      });
      clearInterval(ping); res.end(); return;
    }

    await Promise.all(tasks);

    // final payload
    clearInterval(prog);
    const merged = mergeResults(oai||{}, claude||{});
    const elapsed = Date.now() - start;
    send("progress", { pct: 100 });
    send("final", { ...merged, telemetry: { avg_ms: elapsed, quality: 4.7, tps: 21 } });
  } catch (e) {
    send("final", { consensus: `Error: ${e.message}`, dissents: [], unresolved: [], telemetry: { avg_ms: 0 } });
  } finally {
    clearInterval(ping);
    res.end();
  }
});
