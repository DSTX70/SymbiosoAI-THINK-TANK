export type Token = { name: string; css: string; role: string; }
const rgbVar = (name: string) => getComputedStyle(document.documentElement).getPropertyValue(name).trim()
export const tokenList: Token[] = [
  { name: 'brand.accent.500', css: '--brand-accent-500', role: 'Primary Accent / Links / Buttons' },
  { name: 'text.primary', css: '--text-primary', role: 'Headings' },
  { name: 'text.secondary', css: '--text-secondary', role: 'Body' },
  { name: 'text.tertiary', css: '--text-tertiary', role: 'Captions' },
  { name: 'surface.bg', css: '--surface-bg', role: 'Background' },
  { name: 'surface.panel', css: '--surface-panel', role: 'Panels/Cards' },
  { name: 'surface.elevated', css: '--surface-elevated', role: 'Headers/Chrome' },
  { name: 'border.muted', css: '--border-muted', role: 'Hairlines/Dividers' },
  { name: 'focus.ring', css: '--focus-ring', role: 'Focus Visible' }
]
export function rgbVarToHex (rgb: string) {
  const [r,g,b] = rgb.split(/\s+/).map(v => parseInt(v, 10) || 0);
  return '#' + [r,g,b].map(x => x.toString(16).padStart(2,'0')).join('').toUpperCase();
}
export function getTokenHex(cssVar: string){
  const val = rgbVar(cssVar); return rgbVarToHex(val)
}

cat > src/components/ThemeToggle.tsx <<'TSX'
import React from 'react'
export default function ThemeToggle(){
  const [dark, setDark] = React.useState(document.documentElement.classList.contains('dark'))
  const toggle = () => {
    document.documentElement.classList.toggle('dark')
    const isDark = document.documentElement.classList.contains('dark')
    localStorage.setItem('symbiosoai-theme', isDark ? 'dark' : 'light')
    setDark(isDark)
  }
  return <button onClick={toggle} className="btn btn-primary focus-ring">{dark ? 'Light Mode' : 'Dark Mode'}</button>
}

import { tokenList, getTokenHex } from '@/tokens'
import React from 'react'
function Swatch({cssVar, label}:{cssVar:string,label:string}){
  const color = getComputedStyle(document.documentElement).getPropertyValue(cssVar).trim()
  const hex = getTokenHex(cssVar)
  return (
    <div className="p-4 rounded-xl border border-border-muted bg-surface-elevated">
      <div className="h-16 rounded-md" style={{ backgroundColor: `rgb(${color})` }} />
      <div className="mt-3 text-sm">
        <div className="font-medium">{label}</div>
        <div className="text-text-tertiary">{hex}</div>
      </div>
    </div>
  )
}
export default function ColorTokens(){
  const [, setTick] = React.useState(0)
  React.useEffect(() => {
    const obs = new MutationObserver(() => setTick(t => t+1))
    obs.observe(document.documentElement, { attributes: true, attributeFilter: ['class'] })
    return () => obs.disconnect()
  }, [])
  return (
    <section className="relative rounded-2xl bg-surface-panel shadow-card p-6">
      <h2 className="font-mont text-2xl mb-4">Color Tokens</h2>
      <p className="text-text-secondary mb-6">Use tokens, not hardcoded HEX. Light/Dark mappings are handled via CSS variables.</p>
      <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-4">
        {tokenList.map(t => (<Swatch key={t.name} cssVar={t.css} label={`${t.name} — ${t.role}`} />))}
      </div>
    </section>
  )
}

# SymbiosoAi — Brand Kit (Replit)

A Vite + React + Tailwind mini-site that hosts the live brand handbook:
- Dark/Light token theming (CSS variables)
- Logo showcase
- Color tokens grid
- Typography scale (Montserrat / Inter via Google Fonts)
- Board previews (PNG assets)
- Do/Don't usage rules

## Quick Start
