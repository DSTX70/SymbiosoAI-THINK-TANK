const CLIENT_JS = `
(function(){
  const $=(s,r=document)=>r.querySelector(s);
  const set=(el,html)=>el&&(el.innerHTML=html);

  function connect(url, handlers){
    const es=new EventSource(url);
    es.addEventListener("progress",e=>handlers.onProgress&&handlers.onProgress(JSON.parse(e.data)));
    es.addEventListener("delta",   e=>handlers.onDelta&&handlers.onDelta(JSON.parse(e.data)));
    es.addEventListener("provider",e=>handlers.onProvider&&handlers.onProvider(JSON.parse(e.data)));
    es.addEventListener("final",   e=>{handlers.onFinal&&handlers.onFinal(JSON.parse(e.data)); es.close();});
    es.onerror=()=>{handlers.onError&&handlers.onError("stream error"); es.close();};
    return es;
  }
  function params(base,obj){return base+"?"+new URLSearchParams(obj).toString();}

  const alias = { openai: "analyst", anthropic: "critic" }; // nickname mapping
  const buffers = { simple:{ analyst:"", critic:"" }, guided:{ analyst:"", critic:"" } };
  function appendDelta(prefix, provider, text){
    const nick = alias[provider] || provider;
    buffers[prefix][nick] += text || "";
    const el = $("#"+prefix+"-live-"+nick);
    if(el) el.textContent = buffers[prefix][nick];
  }

  function renderCitations(list){
    if(!list||!list.length) return "<em class='muted'>—</em>";
    return "<ol class='cite-list'>"+list.map((c,i)=>"<li>["+(i+1)+"] "+(c.author?c.author+". ":"")+(c.title?"<strong>“"+c.title+"”</strong>. ":"")+(c.source?c.source+". ":"")+(c.year?c.year+". ":"")+(c.url?"<a href='"+c.url+"' target='_blank' rel='noopener'>"+c.url+"</a>":"")+"</li>").join("")+"</ol>";
  }
  function renderFactChecks(items){
    if(!items||!items.length) return "<em class='muted'>—</em>";
    return "<ul class='cite-list'>"+items.map(fc=>{
      const s=fc.status==="supported"?"<span class='ok'>✓ supported</span>":fc.status==="contradicted"?"<span class='bad'>✕ contradicted</span>":"<span class='muted'>inconclusive</span>";
      return "<li><strong>"+(fc.claim||"")+"</strong> — "+s+(fc.note?(": "+fc.note):"")+"</li>";
    }).join("")+"</ul>";
  }

  function wire(prefix){
    const btn=$("#btn-"+prefix+"-run"); if(!btn) return;
    btn.addEventListener("click", ()=>{
      buffers[prefix].analyst=""; buffers[prefix].critic="";
      // Reset UI
      set($("#"+prefix+"-consensus"),"<em class='muted'>Connecting…</em>");
      ["dissents","unresolved","citations","fc","telemetry"].forEach(id=>set($("#"+prefix+"-"+id),"—"));
      // Reset live sections (collapsed by default)
      const live = $("#"+prefix+"-live");
      if(live){
        live.innerHTML =
          '<details>' +
            '<summary>Analyst (live)</summary>' +
            '<pre id="'+prefix+'-live-analyst"></pre>' +
          '</details>' +
          '<details style="margin-top:10px">' +
            '<summary>Critic (live)</summary>' +
            '<pre id="'+prefix+'-live-critic"></pre>' +
          '</details>';
      }

      const p = $("#"+prefix+"-prompt").value.trim();
      if(!p){ set($("#"+prefix+"-consensus"),"<div class='flash'>Please enter a prompt.</div>"); return; }

      const q = { prompt: p, mode: prefix, temperature: "0.2" };
      if(prefix==="simple"){
        q.require_citations = $("#simple-cite").checked ? "1" : "0";
        q.enable_fact_check = $("#simple-fact").checked ? "1" : "0";
      }else{
        q.selection_mode = $("#sel-mode").value;
        q.response_length = $("#sel-length").value;
        q.turns = $("#inp-rounds").value;
        q.debate_format = $("#sel-format").value;
        q.require_evidence = $("#chk-evidence").checked ? "1" : "0";
        q.require_counterarguments = $("#chk-counter").checked ? "1" : "0";
        q.cite = $("#chk-cite").checked ? "1" : "0";
        q.fact = $("#chk-fact").checked ? "1" : "0";
        q.minsrc = $("#inp-minsrc").value;
      }

      connect(params("/api/think/stream", q), {
        onProgress: d => set($("#"+prefix+"-consensus"),"<em class='muted'>Processing… "+(d.pct||0)+"%</em>"),
        onDelta:    d => appendDelta(prefix, d.provider, d.text),
        onFinal:    d => {
          set($("#"+prefix+"-consensus"), d.consensus || "—");
          set($("#"+prefix+"-dissents"), "<ul>"+(d.dissents||[]).map(x=>"<li>"+(x.position||x)+"</li>").join("")+"</ul>");
          set($("#"+prefix+"-unresolved"), "<ul>"+(d.unresolved||[]).map(u=>"<li>"+u+"</li>").join("")+"</ul>");
          set($("#"+prefix+"-citations"), renderCitations(d.citations));
          set($("#"+prefix+"-fc"), renderFactChecks(d.fact_check?.findings));
          set($("#"+prefix+"-telemetry"), "<p class='muted'>Avg Response: "+(d.telemetry?.avg_ms||"—")+"ms</p>");
        },
        onError:    msg => set($("#"+prefix+"-consensus"), "<div class='flash'>"+msg+"</div>")
      });
    });
  }

  wire("simple");
  wire("guided");
})();
`;

// ---------------- Utilities ----------------
function withTimeout(promise, ms, label="request"){
  return Promise.race([ promise, new Promise((_,rej)=>setTimeout(()=>rej(new Error(`Timeout after ${ms}ms (${label})`)), ms)) ]);
}

async function streamOpenAI(prompt, sys, onDelta){
  if(!ENABLE_OPENAI) return "";
  const stream = await openai.chat.completions.create({
    model: OPENAI_MODEL, temperature: 0.2, stream: true,
    messages: [{role:"system",content:sys},{role:"user",content:String(prompt)}],
  });
  let text = "";
  for await (const chunk of stream) {
    const delta = chunk.choices?.[0]?.delta?.content || "";
    if(delta){ text += delta; onDelta(delta); }
  }
  return text;
}
async function streamAnthropic(prompt, sys, onDelta){
  if(!ENABLE_ANTHROPIC) return "";
  const stream = await anthropic.messages.stream({
    model: CLAUDE_MODEL, max_tokens: 800, temperature: 0.2,
    system: sys, messages:[{role:"user",content:String(prompt)}]
  });
  let text = "";
  stream.on("text",(t)=>{ text += t; onDelta(t); });
  await stream.finalMessage();
  return text;
}

// Structured JSON merge via OpenAI
async function structuredMergeJSON(combined, prompt){
  if(!ENABLE_OPENAI) return null;
  const sys = [
    "Merge analysts’ raw notes into STRICT JSON with keys:",
    "consensus (string), dissents (string[]), unresolved (string[]),",
    "citations (array of {author,title,source,year,url}). No extra keys.",
  ].join(" ");
  const user = `RAW BEGIN<<<
${combined}
RAW END>>>

PROMPT:
${prompt}`;
  const res = await withTimeout(
    openai.chat.completions.create({
      model: OPENAI_MODEL, temperature: 0.1, response_format:{type:"json_object"},
      messages: [{role:"system",content:sys},{role:"user",content:user}]
    }),
    REQ_TIMEOUT_MS, "merge"
  );
  try { return JSON.parse(res.choices?.[0]?.message?.content || "{}"); } catch { return null; }
}

// External verification (optional)
async function verifyClaims(url, key, consensus, dissents, citations){
  if(!url) return { findings: [] };
  try {
    const payload = {
      consensus,
      dissents: (dissents||[]).map(d => (d.position||d)),
      citations
    };
    const headers = {"Content-Type":"application/json"};
    if(key) headers["Authorization"] = "Bearer "+key;
    const resp = await fetch(url, { method:"POST", headers, body: JSON.stringify(payload) });
    if(!resp.ok) throw new Error("verify "+resp.status);
    const data = await resp.json();
    return Array.isArray(data.findings) ? { findings: data.findings } : { findings: [] };
  } catch { return { findings: [] }; }
}

// ---------------- SSE endpoint ----------------
app.get("/api/think/stream", async (req, res) => {
  res.set({"Content-Type":"text/event-stream","Cache-Control":"no-cache","Connection":"keep-alive"});
  const send=(ev,d)=>res.write(`event: ${ev}\ndata: ${JSON.stringify(d)}\n\n`);
  const ka=setInterval(()=>res.write(": ping\n\n"),15000);

  const { prompt="", mode="simple", temperature="0.2" } = req.query;
  const wantCitations = req.query.require_citations === "1" || req.query.cite === "1";
  const wantFact = req.query.enable_fact_check === "1" || req.query.fact === "1";

  if(!prompt.trim()){
    send("final",{consensus:"Please enter a prompt.",dissents:[],unresolved:[],citations:[],fact_check:{findings:[]},telemetry:{avg_ms:0}});
    clearInterval(ka); return res.end();
  }

  let pct=5; const prog=setInterval(()=>{pct=Math.min(92,pct+4);send("progress",{pct});},600);
  const start=Date.now();

  const sysAnalyst = mode==="guided"
    ? "You are the 'Analyst' persona in a multi-model think tank. Stream a well-structured analysis."
    : "You are a precise analyst. Stream your analysis.";
  const sysCritic = mode==="guided"
    ? "You are the 'Critic' persona in a multi-model think tank. Stream a critical perspective."
    : "You are a critical reviewer. Stream your analysis.";

  let oaiText="", claudeText="";
  const tasks=[];

  if(ENABLE_OPENAI){
    tasks.push(
      streamOpenAI(String(prompt), sysAnalyst, (delta)=>send("delta",{provider:"openai",text:delta}))
        .then(t=>send("provider",{provider:"openai",status:"done",len:t.length}))
        .then(()=>{ oaiText = oaiText; }) // noop for clarity
        .catch(e=>send("provider",{provider:"openai",status:"error",message:e.message}))
    );
  }
  if(ENABLE_ANTHROPIC){
    tasks.push(
      streamAnthropic(String(prompt), sysCritic, (delta)=>send("delta",{provider:"anthropic",text:delta}))
        .then(t=>send("provider",{provider:"anthropic",status:"done",len:t.length}))
        .then(()=>{ claudeText = claudeText; }) // noop for clarity
        .catch(e=>send("provider",{provider:"anthropic",status:"error",message:e.message}))
    );
  }

  try{
    // NOTE: capture deltas into strings by intercepting 'delta' events (client accumulates for UI).
    // For server-side merge, we re-stream again quickly or rely on client deltas; for simplicity,
    // re-run fast non-stream calls to get full texts (safer than guessing).
    if(ENABLE_OPENAI){
      tasks.push(
        withTimeout(
          (async ()=>{
            const nonStream = await openai.chat.completions.create({
              model: OPENAI_MODEL, temperature: Number(temperature),
              messages:[{role:"system",content:sysAnalyst},{role:"user",content:String(prompt)}]
            });
            oaiText = nonStream.choices?.[0]?.message?.content || "";
          })(),
          REQ_TIMEOUT_MS, "openai-final"
        )
      );
    }
    if(ENABLE_ANTHROPIC){
      tasks.push(
        withTimeout(
          (async ()=>{
            const msg = await anthropic.messages.create({
              model: CLAUDE_MODEL, max_tokens: 900, temperature: Number(temperature),
              system: sysCritic, messages:[{role:"user",content:String(prompt)}]
            });
            claudeText = msg.content?.[0]?.text || "";
          })(),
          REQ_TIMEOUT_MS, "anthropic-final"
        )
      );
    }

    if(tasks.length===0){
      clearInterval(prog);
      send("final",{consensus:"No AI providers configured.",dissents:[],unresolved:[],citations:[],fact_check:{findings:[]},telemetry:{avg_ms:0}});
      clearInterval(ka); return res.end();
    }

    await Promise.all(tasks);
    clearInterval(prog); send("progress",{pct:100});

    const combined = [oaiText, claudeText].filter(Boolean).join("\n\n");
    // Structured JSON merge (citations included)
    let structured = await structuredMergeJSON(combined, String(prompt));
    if(!structured){
      // minimal fallback
      const pts = (combined.match(/[-–•]\s.+/g)||[]).slice(0,4).map(t=>t.replace(/^[^a-zA-Z0-9]+/,"").trim());
      structured = { consensus: combined.split(/\n{2,}/)[0] || `Summary for “${prompt}”`, dissents: pts.slice(0,3), unresolved:["Clarify assumptions","Define success metrics"], citations: [] };
    }
    const consensus = structured.consensus || `Summary for “${prompt}”`;
    const dissents = (structured.dissents||[]).map(p => (typeof p==="string"? {position:p}:{position:p.position||""}));
    const unresolved = structured.unresolved || [];
    const citations = (wantCitations ? (structured.citations||[]) : []);

    // External fact-check service
    let fact_check = { findings: [] };
    if(wantFact && VERIFY_URL){
      try {
        const payload = {
          consensus,
          dissents: dissents.map(d=>d.position),
          citations
        };
        const headers = {"Content-Type":"application/json"};
        if(VERIFY_API_KEY) headers["Authorization"] = "Bearer "+VERIFY_API_KEY;
        const resp = await fetch(VERIFY_URL, { method:"POST", headers, body: JSON.stringify(payload) });
        if(resp.ok){
          const data = await resp.json();
          if(data && Array.isArray(data.findings)){ fact_check.findings = data.findings; }
        }
      } catch {}
    }

    const elapsed = Date.now() - start;
    send("final",{ consensus, dissents, unresolved, citations, fact_check, telemetry:{avg_ms: elapsed, quality: 4.7, tps: 21} });
  } catch(e){
    send("final",{consensus:`Error: ${e.message}`, dissents:[], unresolved:[], citations:[], fact_check:{findings:[]}, telemetry:{avg_ms:0}});
  } finally{
    clearInterval(ka);
    res.end();
  }
});

  if(!VERIFY_URL) console.log("ℹ️ VERIFY_URL not set — external fact-check disabled");
