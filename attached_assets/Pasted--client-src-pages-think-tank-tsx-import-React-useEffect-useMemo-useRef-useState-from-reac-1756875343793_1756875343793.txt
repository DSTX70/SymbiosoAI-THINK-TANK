/* ============================================================================
   Types aligned to your /api/think schema
============================================================================ */
type ThinkRequest = {
  prompt: string;
  context?: string;
  temperature?: number;
  selected_models?: string[];
  use_case?: string | null;
  require_citations?: boolean;
  enable_fact_check?: boolean;
  debate_title?: string | null;
  timestamp?: string;
  mode?: "simple" | "guided" | "expert";
  selection_mode?: "smart" | "manual" | "domain" | "usecase";
  response_length?: "brief" | "moderate" | "detailed";
  turns?: number;
  debate_format?: "round-robin" | "structured" | "socratic" | "collaborative";
  require_evidence?: boolean;
  require_counterarguments?: boolean;
  verification?: {
    require_citations?: boolean;
    fact_check?: boolean;
    min_sources?: number;
    deep?: boolean;
    forensic?: boolean;
  };
  frameworks?: Array<"systems" | "design" | "first_principles" | "dialectical">;
  max_steps?: number;
  evidence_per_claim?: number;
  include_counterarguments?: boolean;
  ethical_lens?: boolean;
  models?: Array<"analyst" | "pragmatist" | "thoughtful" | "innovator" | "critic">;
  routing?: { analyst?: number; pragmatist?: number; thoughtful?: number; innovator?: number; critic?: number };
  rag?: { enabled?: boolean; top_k?: number; max_tokens?: number; web?: boolean; code?: boolean };
  security?: { pii_redaction?: boolean; log_masking?: boolean; region?: string };
  export_formats?: Array<"pdf" | "json" | "txt" | "story_map">;
};

type ThinkResponse = {
  consensus?: string;
  dissents?: Array<string | { position?: string }>;
  unresolved?: string[];
  telemetry?: { avg_ms?: number; quality?: number; tps?: number; models_used?: string[]; total_tokens?: number };
  echo?: unknown;
};

/* ============================================================================
   Inline: useThink (Cancel + safe timeout + progress ticker)
============================================================================ */
function useThink(options?: { timeoutMs?: number; endpoint?: string }) {
  const timeoutMs = options?.timeoutMs ?? 60_000;     // 60s default
  const endpoint = options?.endpoint ?? "/api/think";

  const controllerRef = useRef<AbortController | null>(null);
  const [isProcessing, setIsProcessing] = useState(false);
  const [processingProgress, setProcessingProgress] = useState(0);

  function startProgress() {
    setProcessingProgress(5);
    let pct = 5;
    const id = window.setInterval(() => {
      pct = Math.min(92, pct + 1); // +1% / 500ms
      setProcessingProgress(pct);
    }, 500);
    return () => window.clearInterval(id);
  }

  async function safeFetch(body: ThinkRequest) {
    // abort any in-flight request first
    controllerRef.current?.abort();
    const controller = new AbortController();
    controllerRef.current = controller;

    const timer = window.setTimeout(() => controller.abort(), timeoutMs);
    const stop = startProgress();
    setIsProcessing(true);

    try {
      const res = await fetch(endpoint, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          temperature: 0.2,
          timestamp: new Date().toISOString(),
          ...body,
        }),
        signal: controller.signal,
      });

      const ct = res.headers.get("content-type") || "";
      if (!res.ok) {
        const text = await res.text().catch(() => "");
        throw new Error(`HTTP ${res.status}: ${text.slice(0, 300)}`);
      }
      if (!ct.includes("application/json")) {
        const text = await res.text();
        throw new Error(`Non-JSON response: ${text.slice(0, 400)}`);
      }

      setProcessingProgress(100);
      return (await res.json()) as ThinkResponse;
    } finally {
      clearTimeout(timer);
      stop();
      setIsProcessing(false);
      if (controllerRef.current === controller) controllerRef.current = null;
      setTimeout(() => setProcessingProgress(0), 800);
    }
  }

  const thinkMutation = useMutation({
    mutationFn: (data: ThinkRequest) => safeFetch(data),
    retry: false, // AI calls: retry = extra tokens & duplicated work
  });

  const cancel = () => controllerRef.current?.abort();

  return { thinkMutation, isProcessing, processingProgress, cancel, setProcessingProgress };
}

/* ============================================================================
   Inline: 280px ThinkToast (non-blocking bottom-right)
============================================================================ */
function ThinkToast(props: { isProcessing: boolean; processingProgress?: number; onCancel?: () => void }) {
  if (!props.isProcessing) return null;
  const pct = Math.min(100, Math.max(0, Math.round(Number(props.processingProgress ?? 0))));
  return (
    <div
      role="status"
      aria-live="polite"
      style={{
        position: "fixed", bottom: 16, right: 16, width: 280,
        background: "#fff", color: "#111827", padding: "12px 14px",
        borderRadius: 10, boxShadow: "0 4px 12px rgba(0,0,0,.15)", zIndex: 99999,
        fontFamily: "Inter, system-ui, sans-serif", fontSize: 14,
      }}
    >
      <div style={{ fontWeight: 600, marginBottom: 4 }}>AI Think Tank Processing…</div>
      <div style={{ fontSize: 12, color: "#6b7280" }}>Collaborative reasoning in progress</div>

      <div style={{ marginTop: 8, height: 6, background: "#eee", borderRadius: 4 }}>
        <div
          style={{
            width: `${pct}%`,
            height: "100%",
            background: "#0099ff",
            borderRadius: 4,
            transition: "width .3s ease",
          }}
        />
      </div>

      <div style={{ marginTop: 10, display: "flex", justifyContent: "flex-end", gap: 8 }}>
        <button
          onClick={props.onCancel}
          aria-label="Cancel processing"
          style={{
            border: "1px solid #e5e7eb", background: "#fff", color: "#111827",
            borderRadius: 6, padding: "6px 10px", fontSize: 12, fontWeight: 600, cursor: "pointer",
          }}
        >
          Cancel
        </button>
      </div>
    </div>
  );
}

/* ============================================================================
   Inline: PromptInput — calls thinkMutation.mutate() directly
============================================================================ */
function PromptInputInline(props: {
  isProcessing: boolean;
  thinkMutation: ReturnType<typeof useThink>["thinkMutation"];
  onSuccess: (data: ThinkResponse) => void;
  onError: (err: unknown) => void;
}) {
  const [prompt, setPrompt] = useState(
    "Should our startup implement a 4-day work week to improve employee satisfaction and productivity?"
  );
  const [estimatedTime] = useState("2-3 minutes");
  const [activeModels] = useState(3);
  const { toast } = useToast();

  const handleStartThinking = () => {
    if (!prompt.trim()) {
      toast({
        title: "Missing Prompt",
        description: "Please enter a question for the think tank to analyze.",
        variant: "destructive",
      });
      return;
    }
    props.thinkMutation.mutate(
      {
        prompt: prompt.trim(),
        mode: "simple",
        temperature: 0.2,
        routing: { analyst: 25, pragmatist: 30, thoughtful: 20, innovator: 15, critic: 10 },
        require_citations: false,
        enable_fact_check: true,
      },
      {
        onSuccess: (data) => {
          toast({ title: "Analysis Complete", description: "AI think tank has finished processing your request." });
          props.onSuccess(data);
        },
        onError: (error) => {
          console.error("Think tank error:", error);
          toast({
            title: "Processing Failed",
            description: "The think tank encountered an issue processing your request.",
            variant: "destructive",
          });
          props.onError(error);
        },
      }
    );
  };

  const handleSaveDraft = () => {
    toast({ title: "Draft Saved", description: "Your prompt has been saved as a draft." });
  };

  return (
    <div className="border-b border-border p-4">
      <div className="w-full space-y-4">
        <div className="flex items-start space-x-4">
          <div className="flex-1">
            <label htmlFor="prompt" className="mb-2 block text-sm font-medium text-foreground">
              What would you like the think tank to analyze?
            </label>
            <Textarea
              id="prompt"
              rows={4}
              className="w-full resize-none rounded-lg border border-border bg-card p-4 text-foreground placeholder-muted-foreground focus:border-transparent focus:ring-2 focus:ring-ring"
              placeholder="Describe your challenge, question, or topic for collaborative AI reasoning..."
              value={prompt}
              onChange={(e) => setPrompt(e.target.value)}
              data-testid="textarea-prompt"
            />
          </div>
        </div>

        <div className="flex items-center justify-between">
          <div className="flex items-center space-x-4">
            <div className="flex items-center space-x-2 text-sm text-muted-foreground">
              <Clock className="h-4 w-4" />
              <span>
                Est. processing: <span data-testid="text-estimated-time">{estimatedTime}</span>
              </span>
            </div>
            <div className="flex items-center space-x-2 text-sm text-muted-foreground">
              <Users className="h-4 w-4" />
              <span>
                <span data-testid="text-active-models">{activeModels}</span> AI models active
              </span>
            </div>
          </div>
          <div className="flex items-center space-x-3">
            <Button
              variant="ghost"
              onClick={handleSaveDraft}
              className="text-muted-foreground hover:text-foreground"
              data-testid="button-save-draft"
            >
              <Save className="mr-2 h-4 w-4" />
              Save Draft
            </Button>
            <Button
              onClick={handleStartThinking}
              disabled={props.isProcessing}
              className="bg-primary px-6 py-2 text-primary-foreground hover:bg-primary/90"
              data-testid="button-start-thinking"
            >
              <Brain className="mr-2 h-4 w-4" />
              {props.isProcessing ? "Processing..." : "Start Thinking"}
            </Button>
          </div>
        </div>
      </div>
    </div>
  );
}

/* ============================================================================
   Page component with responsive GRID layout
   < lg : center only
   lg..xl : left (256px) + center
   xl+    : left (256px) + center + right (288px)
============================================================================ */
export default function ThinkTank() {
  const [isDark, setIsDark] = useState(false);

  const { thinkMutation, isProcessing, processingProgress, cancel, setProcessingProgress } = useThink({
    timeoutMs: 60_000,
    endpoint: "/api/think",
  });

  // theme toggle
  useEffect(() => {
    document.documentElement.classList.toggle("dark", isDark);
  }, [isDark]);

  // belt-and-suspenders progress animation (kept here to ensure UX)
  useEffect(() => {
    if (!isProcessing) return;
    setProcessingProgress(5);
    let pct = 5;
    const id = window.setInterval(() => {
      pct = Math.min(92, pct + 1);
      setProcessingProgress(pct);
    }, 500);
    return () => window.clearInterval(id);
  }, [isProcessing, setProcessingProgress]);

  const toggleTheme = () => setIsDark((d) => !d);

  const currentSession = useMemo(
    () => ({
      mode: "Simple Reasoning",
      models: 3,
      startTime: new Date().toLocaleTimeString("en-US", {
        hour: "numeric",
        minute: "2-digit",
        hour12: true,
      }),
    }),
    []
  );

  return (
    <div className="min-h-screen flex flex-col bg-background text-foreground">
      {/* Toast (non-blocking) */}
      <ThinkToast isProcessing={isProcessing} processingProgress={processingProgress} onCancel={cancel} />

      {/* Header */}
      <header className="bg-card border-b border-border px-6 py-4">
        <div className="flex items-center justify-between">
          <div className="flex items-center space-x-4">
            <div className="flex items-center space-x-2">
              <Brain className="h-5 w-5 text-primary" />
              <h1 className="text-xl font-semibold text-foreground">SymbiosoAi ThinkTank</h1>
            </div>
            <span className="rounded bg-muted px-2 py-1 text-xs font-medium text-muted-foreground">BETA</span>
          </div>
          <div className="flex items-center space-x-4">
            <Button variant="ghost" size="icon" onClick={toggleTheme} data-testid="button-toggle-theme">
              {isDark ? <Sun className="h-4 w-4" /> : <Moon className="h-4 w-4" />}
            </Button>
            <Button variant="ghost" size="icon" data-testid="button-help">
              <HelpCircle className="h-4 w-4" />
            </Button>
            <div className="flex items-center space-x-2 rounded-lg bg-muted px-3 py-2">
              <div className="h-2 w-2 rounded-full bg-secondary" />
              <span className="text-sm text-muted-foreground">API Connected</span>
            </div>
          </div>
        </div>
      </header>

      {/* Main: Responsive GRID — sidebars when space allows */}
      <main
        className="
          min-h-0 flex-1 grid
          grid-cols-[minmax(0,1fr)]                 /* < lg: center only */
          lg:grid-cols-[16rem_minmax(0,1fr)]        /* lg..xl: left + center */
          xl:grid-cols-[16rem_minmax(0,1fr)_18rem]  /* xl+: left + center + right */
        "
      >
        {/* Left sidebar (256px) */}
        <aside className="hidden lg:block border-r border-border overflow-y-auto">
          <div className="w-64">
            <ConfigurationSidebar currentSession={currentSession} onSessionUpdate={() => {}} />
          </div>
        </aside>

        {/* Center (fills remaining width) */}
        <section className="min-w-0 overflow-y-auto px-4 md:px-6">
          <PromptInputInline
            isProcessing={isProcessing}
            thinkMutation={thinkMutation}
            onSuccess={() => {}}
            onError={() => {}}
          />
          <ResultsArea isProcessing={isProcessing} data={thinkMutation.data} error={thinkMutation.error} />
        </section>

        {/* Right sidebar (288px) */}
        <aside className="hidden xl:block border-l border-border overflow-y-auto">
          <div className="w-72">
            <TelemetryPanel data={thinkMutation.data?.telemetry} />
          </div>
        </aside>
      </main>
    </div>
  );
}
