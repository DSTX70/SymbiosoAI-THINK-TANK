  const set = (el, html)=>el&&(el.innerHTML=html);
  function dl(filename, json){
    const blob = new Blob([JSON.stringify(json, null, 2)], {type: "application/json"});
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url; a.download = filename; a.click();
    URL.revokeObjectURL(url);

  let lastFinalSimple = null;

    const es = new EventSource(url);
    const on=(ev,fn)=>es.addEventListener(ev,e=>{ try{ fn(JSON.parse(e.data)); }catch{} });
    on("ready",    d=>handlers.onReady?.(d));
    on("step",     d=>handlers.onStep?.(d));
    on("progress", d=>handlers.onProgress?.(d));
    on("provider", d=>handlers.onProvider?.(d));
    on("delta",    d=>handlers.onDelta?.(d));
    on("final",    d=>{ handlers.onFinal?.(d); es.close(); });
    es.onerror=()=>{ handlers.onError?.("stream error"); es.close(); };
    return es;
  }
  function urlWithParams(path,params){ const usp=new URLSearchParams(params); return path+"?"+usp.toString(); }

  function renderCitations(list){
    if(!list||!list.length)return "<em class='muted'>—</em>";
    return "<ol class='cite-list'>" + list.map((c,i)=>(
      "<li>["+(i+1)+"] "+(c.author?c.author+". ":"")+(c.title?"<strong>“"+c.title+"”</strong>. ":"")+(c.source?c.source+". ":"")+(c.year?c.year+". ":"")+(c.url?"<a href='"+c.url+"' target='_blank' rel='noopener'>"+c.url+"</a>":"")+"</li>"
    )).join("") + "</ol>";
  }
  function renderFactChecks(items){
    if(!items||!items.length)return "<em class='muted'>—</em>";
    return "<ul class='cite-list'>" + items.map(fc=>{
      const status = fc.status==="supported" ? "<span class='ok'>✓ supported</span>" :
                    fc.status==="contradicted" ? "<span class='bad'>✕ contradicted</span>" :
                    "<span class='muted'>inconclusive</span>";
      return "<li><strong>"+(fc.claim||"")+"</strong> — "+status+(fc.note?(": "+fc.note):"")+(fc.citations&&fc.citations.length?(" "+renderCitations(fc.citations)):"")+"</li>";
    }).join("") + "</ul>";
  }
  function renderFinal(d, prefix){

    S("citations").innerHTML  = renderCitations(d.citations);
    S("fc").innerHTML         = renderFactChecks(d.fact_check?.findings);
    const t = d.telemetry || {};
    S("telemetry").innerHTML  = "<p class='muted'>Avg Response: "+(t.avg_ms??"—")+"ms</p><p class='muted'>Quality: "+(t.quality??"—")+"</p><p class='muted'>Tokens/sec: "+(t.tps??"—")+"</p>";
  }

  // SIMPLE
  const sBtn = $("#btn-simple-run");
  const sSave = $("#btn-simple-savejson");
  if(sSave){ sSave.addEventListener("click", ()=> lastFinalSimple && dl("symbiosoai_simple_result.json", lastFinalSimple)); }
  if(sBtn){
    sBtn.addEventListener("click", ()=>{
      const url = urlWithParams("/api/think/stream",{
        prompt: $("#simple-prompt").value.trim(),
        mode: "simple",
        require_citations: $("#simple-cite").checked ? "1":"0",
        enable_fact_check: $("#simple-fact").checked ? "1":"0",
        enable_rag: $("#simple-rag")?.checked ? "1":"0",
        live_web:  $("#simple-web")?.checked ? "1":"0",
        min_sources: $("#simple-minsrc")?.value || "3",
        temperature: "0.2"
      });
      const S=k=>$("#simple-"+k);
      ["consensus","dissents","unresolved","citations","fc","live-openai","live-claude","claims"].forEach(k=>S(k).innerHTML="—");
      lastFinalSimple = null;
      connect(url,{
        onStep:     d => { if(d.step==="claims") S("claims").innerHTML="<ul class='cite-list'>"+d.claims.map(c=>"<li>"+c+"</li>").join("")+"</ul>"; },

        onDelta:    d => {

          if(d.provider==="anthropic"){ if(S("live-claude").innerHTML==="—") S("live-claude").innerHTML=""; S("live-claude").innerHTML += d.text.replace(/</g,"&lt;"); }
        },
        onFinal:    d => { lastFinalSimple = d; renderFinal(d,"simple"); },
        onError:    m => S("consensus").innerHTML="<div class='flash'>"+m+"</div>"
      });
    });
  }

  // GUIDED
  const gBtn = $("#btn-guided-run");
  const gSave = $("#btn-guided-savejson");
  if(gSave){ gSave.addEventListener("click", ()=> lastFinalGuided && dl("symbiosoai_guided_result.json", lastFinalGuided)); }

        require_counterarguments: $("#chk-counter").checked ? "1":"0",
        require_citations: $("#chk-cite").checked ? "1":"0",
        enable_fact_check: $("#chk-fact").checked ? "1":"0",
        enable_rag: $("#guided-rag")?.checked ? "1":"0",
        live_web:  $("#guided-web")?.checked ? "1":"0",
        min_sources: $("#inp-minsrc").value,
        temperature: "0.2"
      });
      const S=k=>$("#guided-"+k);
      ["consensus","dissents","unresolved","citations","fc","live-openai","live-claude","claims"].forEach(k=>S(k).innerHTML="—");
      lastFinalGuided = null;

// ---------- Utilities ----------
async function withTimeout(p, ms, label="request"){
  return Promise.race([ p, new Promise((_,rej)=>setTimeout(()=>rej(new Error(`Timeout after ${ms}ms (${label})`)), ms)) ]);
}
async function streamOpenAI(prompt, sys, temp, onDelta){
  if(!openai) return "";
  const stream = await withTimeout(openai.chat.completions.create({
    model: OPENAI_MODEL, temperature: temp, stream: true,
    messages: [{role:"system",content:sys},{role:"user",content:prompt}]
  }), TIMEOUT_MS,"openai-stream");
  let buf=""; for await (const evt of stream) {
    const d = evt.choices?.[0]?.delta?.content || "";
    if(d){ buf+=d; onDelta(d); }
  } return buf;
}
async function streamAnthropic(prompt, sys, temp, onDelta){
  if(!anthropic) return "";
  const msg = await withTimeout(anthropic.messages.stream({
    model: CLAUDE_MODEL, max_tokens: 1200, temperature: temp, system: sys,
    messages: [{ role:"user", content: prompt }]
  }), TIMEOUT_MS,"anthropic-stream");
  let buf=""; msg.on("text", t => { buf+=t; onDelta(t); }); await msg.done(); return buf;
}
async function extractClaims(prompt){
  if(!openai) return naiveClaims(prompt);
  try{
    const res = await withTimeout(openai.chat.completions.create({
      model: OPENAI_MODEL, temperature:0.2, response_format:{type:"json_object"},
      messages:[
        { role:"system", content:'Extract distinct factual claims. Return JSON: {"claims":["..."]} (max 6).' },
        { role:"user", content: prompt }
      ]
    }), TIMEOUT_MS,"claim-extract");
    const content = res.choices?.[0]?.message?.content || "{\"claims\":[]}";
    const parsed = JSON.parse(content);
    return Array.isArray(parsed.claims) ? parsed.claims.slice(0,6) : naiveClaims(prompt);
  }catch{ return naiveClaims(prompt); }
}
function naiveClaims(prompt){
  return (prompt.match(/[^.?!]+[.?!]/g) || [prompt]).map(s=>s.trim()).filter(Boolean).slice(0,6);
}
async function verifyClaim(claim){
  if(!openai) return { status:"inconclusive", note:"OpenAI not configured" };
  try{
    const res = await withTimeout(openai.chat.completions.create({
      model: OPENAI_MODEL, temperature:0.1, response_format:{type:"json_object"},
      messages:[
        { role:"system", content:'Given a factual claim, label it as supported/contradicted/inconclusive. Return {"status":"...","note":""}.' },
        { role:"user", content: claim }
      ]
    }), TIMEOUT_MS,"verify-claim");
    const content = res.choices?.[0]?.message?.content || "{\"status\":\"inconclusive\",\"note\":\"\"}";
    const parsed = JSON.parse(content);
    return { status: parsed.status || "inconclusive", note: parsed.note || "" };
  }catch{ return { status:"inconclusive", note:"verification failed" }; }
}
async function ragCitations(query){
  if(!ENABLE_RAG) return [];
  try{
    const resp = await withTimeout(fetch(process.env.RAG_SEARCH_URL, {
      method:"POST", headers:{ "Content-Type":"application/json" }, body: JSON.stringify({ q: query })
    }), 12000, "rag");
    if(!resp.ok) return [];
    const json = await resp.json();
    return normalizeCitations(json?.results||[]);
  }catch{ return []; }
}
async function webCitations(query, minSources){
  if(!ENABLE_PERPLEXITY) return [];
  try{
    const resp = await withTimeout(fetch("https://api.perplexity.ai/chat/completions", {
      method:"POST",
      headers:{ "Content-Type":"application/json", "Authorization":"Bearer "+process.env.PERPLEXITY_API_KEY },
      body: JSON.stringify({
        model: PERP_MODEL, temperature: 0.2,
        messages: [
          { role:"system", content:'Return ONLY JSON: {"citations":[{"author":"","title":"","source":"","year":"","url":""},...]}' },
          { role:"user", content:`Find credible references for: ${query}` }
        ]
      })
    }), 20000, "perplexity");
    if(!resp.ok) return [];
    const json = await resp.json();
    const content = json?.choices?.[0]?.message?.content || "{}";
    return normalizeCitations(JSON.parse(content)?.citations||[]).slice(0, Math.max(3, Number(minSources)||3));
  }catch{ return []; }
}
function normalizeCitations(list){
  return list.map(c=>({
    author: c.author || c.authors || "",
    title:  c.title  || "",
    source: c.source || c.publisher || "",
    year:   c.year   || c.date || "",
    url:    c.url    || c.link || ""
  })).filter(x => x.title || x.url);
}
function summarize(oaiText,claudeText){
  const merged=[oaiText,claudeText].filter(Boolean).join("\n\n");
  const consensus = merged ? (merged.split("\n").find(l=>/conclusion|consensus|recommend/i.test(l)) || merged.slice(0,220)) : "No consensus.";
  const bullets   = (merged.match(/(^|\n)[-–•]\s.*?/g)||[]).map(b=>b.replace(/^\s*[-–•]\s?/,'').trim());
  const dissents  = bullets.slice(0,3).map(b=>({position:b}));
  const unresolved= ["Clarify assumptions","Define success metrics"];
  return { consensus,dissents,unresolved };
}

// ---------- SSE Pipeline ----------
app.get("/api/think/stream", async (req,res)=>{
  res.set({"Content-Type":"text/event-stream","Cache-Control":"no-cache, no-transform","Connection":"keep-alive","X-Accel-Buffering":"no"});
  res.flushHeaders && res.flushHeaders();
  const send = (ev,data)=>{res.write(`event: ${ev}\n`);res.write(`data: ${JSON.stringify(data)}\n\n`);};
  const ping = setInterval(()=>res.write(": ping\n\n"), 15000);

  const {
    prompt="", mode="simple", temperature="0.2",
    require_citations="0", enable_fact_check="0",
    enable_rag="0", live_web="0", min_sources="3"
  } = req.query;

  if(!String(prompt).trim()){
    send("final",{consensus:"Please enter a prompt.",dissents:[],unresolved:[],citations:[],fact_check:{findings:[]},telemetry:{avg_ms:0}});
    clearInterval(ping); res.end(); return;
  }

  try{
    send("ready",{ok:true});
    let pct=5; const prog=setInterval(()=>{pct=Math.min(92,pct+3);send("progress",{pct});},600);
    const t0=Date.now();

    // 1) Claims
    const claims = await extractClaims(String(prompt));
    send("step",{step:"claims", claims});

    // 2) Streams (OpenAI/Claude)
    let oaiText="", claudeText="";
    const sysA = mode==="guided" ? "You are the 'Analyst' persona in a multi-model think tank. Think rigorously. Reply as full paragraphs." : "You are a precise analyst. Reply as full paragraphs.";
    const sysC = mode==="guided" ? "You are the 'Critic' persona in a multi-model think tank. Challenge assumptions briefly. Reply as full paragraphs." : "You are a critical reviewer. Reply as full paragraphs.";

    const tasks=[];
    if(ENABLE_OPENAI){
      tasks.push((async()=>{try{
        oaiText = await streamOpenAI(String(prompt), sysA, Number(temperature), (d)=>send("delta",{provider:"openai",text:d}));
        send("provider",{provider:"openai",status:"done"});
      }catch(e){send("provider",{provider:"openai",status:"error",message:e.message});}})());
    }
    if(ENABLE_ANTHROPIC){
      tasks.push((async()=>{try{
        claudeText = await streamAnthropic(String(prompt), sysC, Number(temperature), (d)=>send("delta",{provider:"anthropic",text:d}));
        send("provider",{provider:"anthropic",status:"done"});
      }catch(e){send("provider",{provider:"anthropic",status:"error",message:e.message});}})());
    }
    if(tasks.length===0){await new Promise(r=>setTimeout(r,800));oaiText="No AI providers configured. Set OPENAI_API_KEY and/or ANTHROPIC_API_KEY.";}
    else{await Promise.all(tasks);}

    // 3) Verification passes per claim
    let fcFindings=[];
    if(enable_fact_check==="1" && claims.length){
      fcFindings = await Promise.all(claims.map(async c=>{
        const v = await verifyClaim(c);
        return { claim:c, status:v.status, note:v.note, citations:[] };
      }));
    }

    // 4) Gather citations: RAG → Web fallback → placeholder if required
    let citations=[];
    if(enable_rag==="1"){ citations = citations.concat(await ragCitations(String(prompt))); }
    if(live_web==="1"){ citations = citations.concat(await webCitations(String(prompt), Number(min_sources)||3)); }
    if(require_citations==="1" && citations.length===0){
      citations = [{ author:"Symbioso Research", title:"Placeholder reference (enable RAG/Web)", source:"Internal", year:"2025", url:"" }];
    }
    // Attach a citation per finding (round-robin)
    if(fcFindings.length && citations.length){
      for(let i=0;i<fcFindings.length;i++){ fcFindings[i].citations = [ citations[i % citations.length] ]; }
    }

    // 5) Summarize to final
    const merged = summarize(oaiText, claudeText);
    const elapsed = Date.now() - t0;

    clearInterval(prog);
    send("progress",{pct:100});
    send("final",{ ...merged, citations, fact_check:{findings:fcFindings}, telemetry:{avg_ms:elapsed,quality:4.7,tps:21} });
  }catch(e){
    send("final",{consensus:`Error: ${e.message}`,dissents:[],unresolved:[],citations:[],fact_check:{findings:[]},telemetry:{avg_ms:0}});
  }finally{
    clearInterval(ping);
    res.end();
  }
});
