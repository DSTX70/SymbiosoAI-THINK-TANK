// --------------- CLIENT JS (SSE streaming) ----------------
const CLIENT_JS = `
(function(){
  const $ = (s,r=document)=>r.querySelector(s);
  const set = (el,html)=>el&&(el.innerHTML=html);

  function connect(url, handlers){
    const es = new EventSource(url);
    es.addEventListener("progress", e=>handlers.onProgress && handlers.onProgress(JSON.parse(e.data)));
    es.addEventListener("final", e=>{handlers.onFinal && handlers.onFinal(JSON.parse(e.data)); es.close();});
    es.onerror=()=>{handlers.onError && handlers.onError("stream error"); es.close();}
    return es;
  }
  function urlWithParams(base, params){return base+"?"+new URLSearchParams(params).toString();}

  function renderCitations(list){
    if(!list||!list.length) return "<em class='muted'>—</em>";
    return "<ol class='cite-list'>"+list.map((c,i)=>
      "<li>["+ (i+1) +"] "+ (c.author?c.author+". ":"") +
      (c.title?"<strong>“"+c.title+"”</strong>. ":"") +
      (c.source?c.source+". ":"") +
      (c.year?c.year+". ":"") +
      (c.url?"<a href='"+c.url+"' target='_blank'>"+c.url+"</a>":"") +"</li>"
    ).join("")+"</ol>";
  }
  function renderFactChecks(items){
    if(!items||!items.length) return "<em class='muted'>—</em>";
    return "<ul class='cite-list'>"+items.map(fc=>{
      const status = fc.status==="supported"?"<span class='ok'>✓ supported</span>":
                    fc.status==="contradicted"?"<span class='bad'>✕ contradicted</span>":
                    "<span class='muted'>inconclusive</span>";
      return "<li><strong>"+(fc.claim||"")+"</strong> — "+status+(fc.note?(": "+fc.note):"")+"</li>";
    }).join("")+"</ul>";
  }

  function wire(btnId, prefix){
    const btn=$(btnId); if(!btn)return;
    btn.addEventListener("click",()=>{
      const url=urlWithParams("/api/think/stream",{
        prompt: $("#"+prefix+"-prompt").value,
        mode: prefix,
        require_citations: $("#"+prefix+"-cite").checked?"1":"0",
        enable_fact_check: $("#"+prefix+"-fact").checked?"1":"0"
      });
      set($("#"+prefix+"-consensus"),"<em class='muted'>Connecting…</em>");
      connect(url,{
        onProgress:d=>set($("#"+prefix+"-consensus"),"<em class='muted'>Processing… "+(d.pct||0)+"%</em>"),
        onFinal:d=>{
          set($("#"+prefix+"-consensus"),d.consensus||"—");
          set($("#"+prefix+"-dissents"),"<ul>"+(d.dissents||[]).map(x=>"<li>"+(x.position||x)+"</li>").join("")+"</ul>");
          set($("#"+prefix+"-unresolved"),"<ul>"+(d.unresolved||[]).map(u=>"<li>"+u+"</li>").join("")+"</ul>");
          set($("#"+prefix+"-citations"),renderCitations(d.citations));
          set($("#"+prefix+"-fc"),renderFactChecks(d.fact_check?.findings));
          set($("#"+prefix+"-telemetry"),"<p class='muted'>Avg Response: "+(d.telemetry?.avg_ms||"—")+"ms</p>");
        },
        onError:msg=>set($("#"+prefix+"-consensus"),"<div class='flash'>"+msg+"</div>")
      });
    });
  }
  wire("#btn-simple-run","simple");
  wire("#btn-guided-run","guided");
})();
`;

// --------------- SSE Endpoint ----------------
app.get("/api/think/stream",(req,res)=>{
  res.set({"Content-Type":"text/event-stream","Cache-Control":"no-cache","Connection":"keep-alive"});
  const send=(ev,d)=>res.write(`event: ${ev}\ndata: ${JSON.stringify(d)}\n\n`);
  let pct=5;const prog=setInterval(()=>{pct=Math.min(92,pct+5);send("progress",{pct});},500);
  setTimeout(()=>{clearInterval(prog);
    send("progress",{pct:100});
    // Example structured result
    send("final",{
      consensus:"Consensus for '"+(req.query.prompt||"your prompt")+"'",
      dissents:[{position:"Option A not viable"},{position:"Option B costly"}],
      unresolved:["Need more data","Check timeline"],
      citations:req.query.require_citations==="1"?[
        {author:"HBR",title:"Compressed Schedules",source:"Harvard Business Review",year:"2023",url:"https://example.com/hbr"},
        {author:"McKinsey",title:"Shift Coverage",source:"McKinsey",year:"2022",url:"https://example.com/mck"}
      ]:[],
      fact_check:req.query.enable_fact_check==="1"?{
        findings:[
          {claim:"4-day week reduces attrition",status:"supported",note:"Strong evidence"},
          {claim:"Always increases burnout",status:"contradicted",note:"Evidence is mixed"}
        ]
      }:{findings:[]},
      telemetry:{avg_ms:1234,quality:4.7,tps:21}
    });
    res.end();
  },3000);
});
