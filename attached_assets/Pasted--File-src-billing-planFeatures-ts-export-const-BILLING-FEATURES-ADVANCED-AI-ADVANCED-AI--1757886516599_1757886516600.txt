// File: src/middleware/requireFeature.ts
import type { Request, Response, NextFunction } from 'express';
import { BILLING_FEATURES, PLAN_FEATURES, type BillingFeature } from '../billing/planFeatures';
import { hasFeatureAccess } from '../services/entitlements';

export function requireFeature(feature: BillingFeature) {
  return async (req: Request, res: Response, next: NextFunction) => {
    try {
      if (!req.user) {
        return res.status(401).json({ error: 'Authentication required', code: 'AUTH_REQUIRED' });
      }

      const isDemo = Boolean((req.user as any).isDemo);
      const userSub = ((req.user as any).subscription ?? {}) as { plan?: string };
      const plan = userSub.plan || (isDemo ? 'demo' : 'free');

      const workspaceId = (req.params as any).workspaceId || (req.body as any).workspaceId || (req.query as any).workspaceId;

      if (feature === BILLING_FEATURES.ADVANCED_AI && PLAN_FEATURES[plan]?.includes(BILLING_FEATURES.ADVANCED_AI)) {
        return next();
      }

      if (!workspaceId) {
        if (isDemo) {
          // Auto-create ephemeral demo workspace for demo users
          const demoWorkspaceId = 'demo-workspace-' + (req.user as any).id;
          (req as any).workspaceId = demoWorkspaceId;
          return next();
        }
        return res.status(400).json({ error: 'Workspace context required for feature access', code: 'WORKSPACE_CONTEXT_REQUIRED' });
      }

      const hasAccessFlag = await hasFeatureAccess(workspaceId, feature);
      if (!hasAccessFlag) {
        return res.status(402).json({ error: 'Feature requires upgrade', code: 'FEATURE_REQUIRES_UPGRADE', feature, workspaceId });
      }

      return next();
    } catch (err) {
      console.error('Feature requirement check failed:', err);
      return res.status(500).json({ error: 'Internal server error', code: 'INTERNAL_ERROR' });
    }
  };
}

export interface WorkspaceRow {
  id: string;
  slug?: string;
  name: string;
  ownerUserId: string;
  isDemo: boolean;
  seeded?: boolean;
}

export interface DemoDocRow {
  id: string;
  workspaceId: string;
  title: string;
  content: string;
  createdAt: string;
}

class InMemoryStorage {
  private workspaces = new Map<string, WorkspaceRow>();
  private docs = new Map<string, DemoDocRow[]>();

  async getWorkspace(id: string): Promise<WorkspaceRow | null> {
    return this.workspaces.get(id) ?? null;
  }

  async getWorkspaceBySlugOrOwner(opts: { ownerUserId: string; slug: string }): Promise<WorkspaceRow | null> {
    return Array.from(this.workspaces.values()).find(
      (w) => w.ownerUserId === opts.ownerUserId && (w.id === opts.slug || w.slug === opts.slug)
    ) ?? null;
  }

  async createWorkspace(ws: WorkspaceRow): Promise<void> {
    this.workspaces.set(ws.id, ws);
  }

  async updateWorkspace(id: string, patch: Partial<WorkspaceRow>): Promise<void> {
    const current = this.workspaces.get(id);
    if (current) {
      this.workspaces.set(id, { ...current, ...patch });
    }
  }

  async upsertDemoDocs(workspaceId: string, docs: DemoDocRow[]): Promise<void> {
    this.docs.set(workspaceId, docs);
  }

  async getDemoDocs(workspaceId: string): Promise<DemoDocRow[]> {
    return this.docs.get(workspaceId) ?? [];
  }
}

export const storage = new InMemoryStorage();

/**
 * If caller is a demo user and no workspaceId is supplied, auto-create/attach a demo workspace.
 * Also: if demo user has upgraded to non-demo plan, purge any demo workspaces/docs to avoid leakage.
 */
export async function ensureWorkspaceForDemo(req: Request, _res: Response, next: NextFunction) {
  try {
    const user: any = req.user;
    const hasWs = (req.params as any).workspaceId || (req.body as any).workspaceId || (req.query as any).workspaceId;

    // Guard: if user was demo but is now on a real plan, purge demo ws/docs.
    if (user?.isDemo && user?.subscription?.plan && user.subscription.plan !== 'demo') {
      await storage.deleteDemoWorkspacesAndDocs?.(user.id);
      user.isDemo = false; // flip flag so downstream respects real plan
    }

    if (user?.isDemo && !hasWs) {
      const ws = await ensureDemoWorkspace(user.id);
      (req.body as any) = { ...(req.body as any), workspaceId: ws.id };
      (req as any).workspace = ws;
    }
    return next();
  } catch (e) {
    // eslint-disable-next-line no-console
    console.error('ensureWorkspaceForDemo error:', e);
    return next();
  }
}

import { Pool, PoolClient } from 'pg';

export type UUID = string;

export interface UserRow {

export interface WorkspaceRow {
  id: string; // allow slug-like id `demo-ws-<userId>`
  name: string;
  ownerUserId: UUID;
  isDemo: boolean;
  seeded: boolean;
  createdAt: Date;
  updatedAt: Date;
}

export interface DemoDocRow {
  id: string;
  workspaceId: string;
  title: string;
  content: string;
  createdAt: Date;
}

export interface Storage {
  // Users/session (minimal)
  upsertUser(user: Partial<UserRow> & { id: UUID; email: string }): Promise<void>;
  upsertSession(session: any): Promise<void>;

  // Workspaces
  getWorkspace(id: string): Promise<WorkspaceRow | null>;
  getWorkspaceBySlugOrOwner(arg: { ownerUserId: UUID; slug: string }): Promise<WorkspaceRow | null>;
  createWorkspace(ws: { id: string; name: string; ownerUserId: UUID; isDemo: boolean; seeded?: boolean }): Promise<void>;
  updateWorkspace(id: string, patch: Partial<Pick<WorkspaceRow, 'name' | 'seeded'>>): Promise<void>;

  // Demo docs
  upsertDemoDocs(workspaceId: string, docs: { id: string; title: string; content: string; createdAt: string }[]): Promise<void>;
  getDemoDocs(workspaceId: string): Promise<DemoDocRow[]>;
}

export class PgStorage implements Storage {
  private pool: Pool;

  constructor(config?: ConstructorParameters<typeof Pool>[0]) {
    this.pool = new Pool(config);
  }

  private async tx<T>(fn: (c: PoolClient) => Promise<T>): Promise<T> {
    const client = await this.pool.connect();
    try {
      await client.query('BEGIN');
      const res = await fn(client);
      await client.query('COMMIT');
      return res;
    } catch (e) {
      await client.query('ROLLBACK');
      throw e;
    } finally {
      client.release();
    }
  }

  async upsertUser(user: Partial<UserRow> & { id: UUID; email: string }): Promise<void> {
    await this.pool.query(
      `INSERT INTO users (id, email, first_name, last_name, profile_image_url, role, is_demo, plan)
       VALUES ($1,$2,$3,$4,$5, COALESCE($6,'user'), COALESCE($7,false), COALESCE($8,'free'))
       ON CONFLICT (id) DO UPDATE SET
         email = EXCLUDED.email,
         first_name = EXCLUDED.first_name,
         last_name = EXCLUDED.last_name,
         profile_image_url = EXCLUDED.profile_image_url,
         role = EXCLUDED.role,
         is_demo = EXCLUDED.is_demo,
         plan = EXCLUDED.plan`,
      [
        user.id,
        user.email,
        user.firstName ?? null,
        user.lastName ?? null,
        user.profileImageUrl ?? null,
        (user as any).role ?? 'user',
        (user as any).isDemo ?? false,
        (user as any).subscription?.plan ?? (user as any).plan ?? 'free',
      ]
    );
  }

  async upsertSession(session: any): Promise<void> {
    // No-op stub. Why: session persistence varies by stack; plug your own store here.
    return;
  }

  async getWorkspace(id: string): Promise<WorkspaceRow | null> {
    const { rows } = await this.pool.query(
      `SELECT id, name, owner_user_id as "ownerUserId", is_demo as "isDemo", seeded, created_at as "createdAt", updated_at as "updatedAt"
       FROM workspaces WHERE id = $1`,
      [id]
    );
    return rows[0] ?? null;
  }

  async getWorkspaceBySlugOrOwner(arg: { ownerUserId: UUID; slug: string }): Promise<WorkspaceRow | null> {
    const { rows } = await this.pool.query(
      `SELECT id, name, owner_user_id as "ownerUserId", is_demo as "isDemo", seeded, created_at as "createdAt", updated_at as "updatedAt"
       FROM workspaces WHERE id = $1 OR (owner_user_id = $2 AND is_demo = true) ORDER BY created_at ASC LIMIT 1`,
      [arg.slug, arg.ownerUserId]
    );
    return rows[0] ?? null;
  }

  async createWorkspace(ws: { id: string; name: string; ownerUserId: UUID; isDemo: boolean; seeded?: boolean }): Promise<void> {
    await this.pool.query(
      `INSERT INTO workspaces (id, name, owner_user_id, is_demo, seeded)
       VALUES ($1,$2,$3,$4, COALESCE($5,false)) ON CONFLICT (id) DO NOTHING`,
      [ws.id, ws.name, ws.ownerUserId, ws.isDemo, ws.seeded ?? false]
    );
  }

  async updateWorkspace(id: string, patch: Partial<Pick<WorkspaceRow, 'name' | 'seeded'>>): Promise<void> {
    const fields: string[] = [];
    const values: any[] = [];
    let i = 1;
    if (patch.name !== undefined) { fields.push(`name = $${i++}`); values.push(patch.name); }
    if (patch.seeded !== undefined) { fields.push(`seeded = $${i++}`); values.push(patch.seeded); }
    if (!fields.length) return;
    values.push(id);
    await this.pool.query(`UPDATE workspaces SET ${fields.join(', ')}, updated_at = NOW() WHERE id = $${i}`, values);
  }

  async upsertDemoDocs(workspaceId: string, docs: { id: string; title: string; content: string; createdAt: string }[]): Promise<void> {
    await this.tx(async (c) => {
      for (const d of docs) {
        await c.query(
          `INSERT INTO demo_docs (id, workspace_id, title, content, created_at)
           VALUES ($1,$2,$3,$4,$5)
           ON CONFLICT (id) DO NOTHING`,
          [d.id, workspaceId, d.title, d.content, d.createdAt]
        );
      }
    });
  }

  async getDemoDocs(workspaceId: string): Promise<DemoDocRow[]> {
    const { rows } = await this.pool.query(
      `SELECT id, workspace_id as "workspaceId", title, content, created_at as "createdAt"
       FROM demo_docs WHERE workspace_id = $1 ORDER BY created_at ASC`,
      [workspaceId]
    );
    return rows as DemoDocRow[];
  }
}

-- Why: minimal schema for users, workspaces, demo docs. Idempotent-ish via IF NOT EXISTS.
CREATE TABLE IF NOT EXISTS users (
  id UUID PRIMARY KEY,
  email TEXT NOT NULL UNIQUE,
  first_name TEXT,
  last_name TEXT,
  profile_image_url TEXT,
  role TEXT NOT NULL DEFAULT 'user',
  is_demo BOOLEAN NOT NULL DEFAULT false,
  plan TEXT NOT NULL DEFAULT 'free',
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS workspaces (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  owner_user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  is_demo BOOLEAN NOT NULL DEFAULT false,
  seeded BOOLEAN NOT NULL DEFAULT false,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS demo_docs (
  id TEXT PRIMARY KEY,
  workspace_id TEXT NOT NULL REFERENCES workspaces(id) ON DELETE CASCADE,
  title TEXT NOT NULL,
  content TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_workspaces_owner ON workspaces(owner_user_id);
CREATE INDEX IF NOT EXISTS idx_demo_docs_ws ON demo_docs(workspace_id);

-- Trigger to update updated_at on workspaces
CREATE OR REPLACE FUNCTION set_updated_at() RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END; $$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_ws_updated ON workspaces;
CREATE TRIGGER trg_ws_updated BEFORE UPDATE ON workspaces
FOR EACH ROW EXECUTE FUNCTION set_updated_at();
